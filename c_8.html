1. Միաչափ զանգված (1D)

int arr[5];

Զանգվածը նույնատիպ տվյալների հավաքածու է։ C++-ում դասիչները (ինդեքսները) միշտ սկսվում են 0-ից։ Սեղմեք ցանկացած վանդակի վրա՝ արժեքը տեսնելու կամ փոխելու համար։

arr[2] = 30; // 3-րդ տարրը
Սեղմեք վանդակի վրա, ապա փորձեք ավելացնել +1, ինչպես դասագրքի օրինակում։

2. Երկչափ զանգված / Մատրից (2D)

int A[3][4];

Առաջին դասիչը ցույց է տալիս տողի համարը, երկրորդը՝ սյան համարը՝ A[տող][սյուն]։ Սեղմեք վանդակի վրա՝ դասիչը տեսնելու համար։

A[0][0] == 12 (Տող 0, Սյուն 0) Ընդամենը՝ 3 × 4 = 12 տարր

Համառոտ ուղեցույց (Թեմա 8)

Հայտարարում և սկզբնարժեքավորում
int arr[4] = {12, 6, 4, 2};
float temp[7];
int m[3][2] = {{1,2},{4,5},{7,8}};
Տարրին դիմելը և փոխելը
arr[2] = 35; // 3-րդ տարրը
cout << grades[2][0];
// 3-րդ տող, 1-ին սյուն
Արտածում կրկնակի ցիկլով
for(int i=0; i<rows; i++) {
  for(int j=0; j<cols; j++)
    cout << a[i][j] << " ";
  cout << endl;
}
